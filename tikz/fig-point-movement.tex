\documentclass[border=3pt,tikz]{standalone}
\usepackage{bm}
\usetikzlibrary{patterns,decorations.pathmorphing}

\begin{document}

\begin{tikzpicture}
    \def\angXzero{0}
    \def\angXone{45}
    
    \coordinate (A)  at (-1,0);
    \coordinate (Xzero) at (1,1);
    \coordinate (Xone) at (7,-2);
    \coordinate (B)  at (9,2);

    \draw[dashed] (A) to[out=45, in=180+\angXzero] (Xzero) to[out=\angXzero,in=180+\angXone] (Xone) to[out=\angXone,in=180] (B);

    \draw[very thick] (Xzero) to[out=\angXzero,in=180+\angXone] (Xone);

    
    \fill (Xzero) circle (2pt) node[below=3pt] {$\bm{y}_0$};
    \draw[-latex, red, thick] (Xzero) -- ++(\angXzero:2.5) 
        node[right, sloped, pos=1] {$\bm{v}(\bm{y}_0,t_0)$};

    \fill (Xone) circle (2pt) node[below right] {$\bm{y}(t)$};
    \draw[-latex, red, thick] (Xone) -- ++(\angXone:1.5) 
        node[right, sloped, pos=1] {$\bm{v}(\bm{y}(t),t)$};
                
\end{tikzpicture}
